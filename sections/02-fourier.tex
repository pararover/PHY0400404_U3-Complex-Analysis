\section{Fourier series}
    Given a periodic function $f(x)$ with a period of $2\pi$ and is integrable over a period, then it can be represented as the trigonometric series
    \begin{align*}
        f(x)=a_0+\sum_{n=1}^{\infty}{\qty(a_n\cos{nx}+b_n\sin{nx})}
    \end{align*}
    In the above equation, the function $f(x)$ is expanded as a Fourier series where the coefficients are given by the Euler formulae:
    \begin{align*}
        a_0&=\dfrac{1}{2\pi}\int\limits_{-\pi}^{\pi}{f(x)\,\dd{x}} \\
        a_n&=\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\cos{nx}\,\dd{x}},\quad n=1,2,3,\dots \\
        b_n&=\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\sin{nx}\,\dd{x}},\quad n=1,2,3,\dots
    \end{align*}
    Note: The Fourier series expansion of a given function exists provided it satisfies the \textit{Dirichlet conditions}.

    \subsection{Dirichlet conditions}
        The particular conditions that a given function $f(x)$, which is periodic over some $\lambda\in\mathbb{R}$, must fulfil in order that it may be expanded as a Fourier series are known as the \textit{Dirichlet conditions}, and may be summarised by the following:
        \begin{enumerate}[I.]
            \item The periodic function $f(x)$ has a finite number of maxima and minima over the range of the period.
            \item The periodic function $f(x)$ has a finite number of discontinuities, of finite values, over the range of the period.
            \item The period function $f(x)$ is absolutely integrable over the range of the period, i.e.
            $$\int\limits_{x}^{x+\lambda}{|f(x)|\,\dd{x}}<\infty$$
        \end{enumerate}

    \subsection{Determination of Fourier coefficients}
        When the Dirichlet conditions are satisfied, a periodic function $f(x)$ can be written as
        \begin{align}
            f(x)=a_0+\sum_{n=1}^{\infty}{\qty(a_n\cos{nx}+b_n\sin{nx})} \label{eqn:fourier-series}
        \end{align}

        \subsubsection{Determination of the constant term $a_0$}
            Integrating equation (\ref{eqn:fourier-series}) on both sides between $-\pi$ and $\pi$, we get
            \begin{align}
                \int\limits_{-\pi}^{\pi}{f(x)\,\dd{x}}&=\int\limits_{-\pi}^{\pi}{\qty[a_0+\sum_{n=1}^{\infty}{\qty(a_n\cos{nx}+b_n\sin{nx})}]\,\dd{x}} \nonumber\\
                    &=a_0\int\limits_{-\pi}^{\pi}{\dd{x}}+\sum_{n=1}^{\infty}{a_n\underbrace{\int\limits_{-\pi}^{\pi}{\cos{nx}\,\dd{x}}}_{=0}}+\sum_{n=1}^{\infty}{b_n\underbrace{\int\limits_{-\pi}^{\pi}{\sin{nx}\,\dd{x}}}_{=0}} \nonumber\\
                \implies a_0\int\limits_{-\pi}^{\pi}{\dd{x}}&=\int\limits_{-\pi}^{\pi}{f(x)\,\dd{x}} \nonumber\\
                \implies a_0\left.x\right|_{-\pi}^{\pi}&=\int\limits_{-\pi}^{\pi}{f(x)\,\dd{x}} \nonumber\\
                \implies a_0\qty[\pi-(-\pi)]&=\int\limits_{-\pi}^{\pi}{f(x)\,\dd{x}} \nonumber\\
                \implies 2\pi a_0&=\int\limits_{-\pi}^{\pi}{f(x)\,\dd{x}} \nonumber\\
                \implies\Aboxed{a_0&=\dfrac{1}{2\pi}\int\limits_{-\pi}^{\pi}{f(x)\,\dd{x}}} \label{eqn:fourier-a0}
            \end{align}

        \subsubsection{Determination of the $a_n$ coefficients}
            Multiplying both sides of equation (\ref{eqn:fourier-series}) by $\cos{mx}$ (where $m$ is any fixed positive integer), we get
            \begin{align*}
                f(x)\cos{mx}=a_0\cos{mx}+\sum_{n=1}^{\infty}{\qty(a_n\cos{nx}\cos{mx}+b_n\sin{nx}\cos{mx})}
            \end{align*}
            Now, integrating the above on both sides between $-\pi$ and $\pi$, we get
            \begin{align}
                \int\limits_{-\pi}^{\pi}{f(x)\cos{mx}\,\dd{x}}&=a_0\underbrace{\int\limits_{-\pi}^{\pi}{\cos{mx}\,\dd{x}}}_{=0}+\sum_{n=1}^{\infty}{a_n\underbrace{\int\limits_{-\pi}^{\pi}{\cos{nx}\cos{mx}\,\dd{x}}}_{=\pi\delta_{nm}}}+\sum_{n=1}^{\infty}{b_n\underbrace{\int\limits_{-\pi}^{\pi}{\sin{nx}\cos{mx}\,\dd{x}}}_{=0}} \nonumber\\
                \implies\sum_{n=1}^{\infty}{\pi a_n\delta_{nm}}=\pi a_n&=\int\limits_{-\pi}^{\pi}{f(x)\cos{nx}\,\dd{x}} \nonumber\\
                \implies\Aboxed{a_n&=\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\cos{nx}\,\dd{x}};\quad n=1,2,3,\dots} \label{eqn:fourier-an}
            \end{align}

        \subsubsection{Determination of the $b_n$ coefficients}
            Multiplying both sides of equation (\ref{eqn:fourier-series}) by $\sin{mx}$ (where $m$ is any fixed positive integer), we get
            \begin{align*}
                f(x)\sin{mx}=a_0\sin{mx}+\sum_{n=1}^{\infty}{\qty(a_n\cos{nx}\sin{mx}+b_n\sin{nx}\sin{mx})}
            \end{align*}
            Now, integrating the above on both sides between $-\pi$ and $\pi$, we get
            \begin{align}
                \int\limits_{-\pi}^{\pi}{f(x)\sin{mx}\,\dd{x}}&=a_0\underbrace{\int\limits_{-\pi}^{\pi}{\sin{mx}\,\dd{x}}}_{=0}+\sum_{n=1}^{\infty}{a_n\underbrace{\int\limits_{-\pi}^{\pi}{\cos{nx}\sin{mx}\,\dd{x}}}_{=0}}+\sum_{n=1}^{\infty}{b_n\underbrace{\int\limits_{-\pi}^{\pi}{\sin{nx}\sin{mx}\,\dd{x}}}_{=\pi\delta_{nm}}} \nonumber\\
                \implies\sum_{n=1}^{\infty}{\pi b_n\delta_{nm}}=\pi b_n&=\int\limits_{-\pi}^{\pi}{f(x)\sin{nx}\,\dd{x}} \nonumber\\
                \implies\Aboxed{b_n&=\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\sin{nx}\,\dd{x}};\quad n=1,2,3,\dots} \label{eqn:fourier-bn}
            \end{align}

        Therefore, we obtain the Fourier coefficients to be:
        \begin{align*}
            a_0&=\dfrac{1}{2\pi}\int\limits_{-\pi}^{\pi}{f(x)\,\dd{x}} \\
            a_n&=\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\cos{nx}\,\dd{x}};\quad n=1,2,3,\dots \\
            b_n&=\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\sin{nx}\,\dd{x}};\quad n=1,2,3,\dots
        \end{align*}
