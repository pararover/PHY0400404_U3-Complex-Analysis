\section{Complex representation of Fourier series}
    Given a periodic function $f(x)$ with period $2\pi$ and satisfying the Dirichlet conditions, its Fourier series expansion is
    \begin{align}
        f(x)=a_0+\sum_{n=1}^{\infty}{\qty(a_n\cos{nx}+b_n\sin{nx})} \label{eqn:fourier-complex}
    \end{align}

    The Euler formula for complex numbers is
    $$e^{i\theta}=\cos{\theta}+i\sin{\theta}$$

    Hence, we can write
    \begin{align}
        e^{inx}&=\cos{nx}+i\sin{nx} \label{eqn:euler+inx} \\
        e^{-inx}&=\cos{nx}-i\sin{nx} \label{eqn:euler-inx}
    \end{align}

    Adding equations (\ref{eqn:euler+inx}) and (\ref{eqn:euler-inx}) gives us
    \begin{align}
        \cos{nx}&=\dfrac{1}{2}\qty(e^{inx}+e^{-inx}) \label{eqn:euler:cosnx}
    \end{align}

    Subtracting equation (\ref{eqn:euler-inx}) from equation (\ref{eqn:euler+inx}) gives us
    \begin{align}
        \sin{nx}&=\dfrac{1}{2i}\qty(e^{inx}-e^{-inx}) \label{eqn:euler:sinnx}
    \end{align}

    Then we have
    \begin{align}
        a_n\cos{nx}+b_n\sin{nx}&=\dfrac{a_n}{2}\qty(e^{inx}+e^{-inx})+\dfrac{b_n}{2i}\qty(e^{inx}-e^{-inx})\qq{[using equations (\ref{eqn:euler:cosnx}) and (\ref{eqn:euler:sinnx})]} \nonumber\\
            &=\dfrac{a_n}{2}\qty(e^{inx}+e^{-inx})-\dfrac{ib_n}{2}\qty(e^{inx}-e^{-inx}) \nonumber \\
            &=\underbrace{\dfrac{1}{2}\qty(a_n-ib_n)}_{=c_n}e^{inx}+\underbrace{\dfrac{1}{2}\qty(a_n+ib_n)}_{=d_n}e^{-inx} \nonumber \\
        \implies &a_n\cos{nx}+b_n\sin{nx}=c_ne^{inx}+d_ne^{-inx} \label{eqn:euler:an-bn-cn-dn}
    \end{align}
    where $c_n=\dfrac{1}{2}\qty(a_n-ib_n)$, $d_n=\dfrac{1}{2}\qty(a_n+ib_n)$ and we let $c_0=a_0$.

    Using equation (\ref{eqn:euler:an-bn-cn-dn}) in equation (\ref{eqn:fourier-complex}), we obtain
    \begin{align}
        f(x)=c_0+\sum_{n=1}^{\infty}{\qty(c_ne^{inx}+d_ne^{-inx})} \label{eqn:fourier-complex-sum}
    \end{align}
    where the $c_n$ coefficients are as follows
    \begin{align}
        c_n&=\dfrac{1}{2}a_n-\dfrac{i}{2}b_n \nonumber\\
            &=\dfrac{1}{2}\qty[\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\cos{nx}\,\dd{x}}]-\dfrac{i}{2}\qty[\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\sin{nx}\,\dd{x}}] \nonumber\\
            &=\dfrac{1}{2\pi}\int\limits_{-\pi}^{\pi}{f(x)\qty(\cos{nx}-i\sin{nx})\,\dd{x}} \nonumber\\
        \implies c_n&=\dfrac{1}{2\pi}\int\limits_{-\pi}^{\pi}{f(x)e^{-inx}\,\dd{x}} \label{eqn:fourier-complex:cn}
    \end{align}
    Also, the $d_n$ coefficients are as follows:
    \begin{align}
        d_n&=\dfrac{1}{2}a_n+\dfrac{i}{2}b_n \nonumber\\
            &=\dfrac{1}{2}\qty[\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\cos{nx}\,\dd{x}}]+\dfrac{i}{2}\qty[\dfrac{1}{\pi}\int\limits_{-\pi}^{\pi}{f(x)\sin{nx}\,\dd{x}}] \nonumber\\
            &=\dfrac{1}{2\pi}\int\limits_{-\pi}^{\pi}{f(x)\qty(\cos{nx}+i\sin{nx})\,\dd{x}} \nonumber\\
        \implies d_n&=\dfrac{1}{2\pi}\int\limits_{-\pi}^{\pi}{f(x)e^{inx}\,\dd{x}} \label{eqn:fourier-complex:dn}
    \end{align}

    Equations (\ref{eqn:fourier-complex:cn}) and (\ref{eqn:fourier-complex:dn}) may be combined by writing $d_n=c_{-n}$.
    
    Therefore, we obtain the complex representation of a Fourier series as
    \begin{align*}
        f(x)=\sum_{n=-\infty}^{\infty}{c_ne^{inx}}
    \end{align*}
    where
    \begin{align*}
        c_n=\dfrac{1}{2\pi}\int\limits_{-\pi}^{\pi}{f(x)e^{-inx}\,\dd{x}};\quad n=0,\pm1,\pm2,\dots
    \end{align*}
